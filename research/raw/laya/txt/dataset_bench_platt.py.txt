"""Fit a threshold / Platt scaling on half the benchmark, score the other half.

Temperature scaling is softmax(z/T): multiplicative, no bias term, so it can push
confidence toward 0.5 but never across it. When a model ranks well but sits
entirely on one side of the boundary - exactly our case here - only a bias term
can fix it. Platt is sigma(a*logit + b).

Reported honestly: Jev's published 62.6% is its RAW output, so the raw column is
the like-for-like comparison and the fitted column is what is reachable with a
calibration slice.
"""
import json, argparse
import numpy as np
from datasets import load_dataset
import laya

ap = argparse.ArgumentParser()
ap.add_argument("models", nargs="+")
args = ap.parse_args()

ds = load_dataset("AreLit/PhishNChips", "emails", split="core")
rows = []
for r in ds:
    try: state = json.loads(r["email_content"])
    except Exception: state = r["email_content"]
    rows.append((state, int(r["phish_label"])))

Q = {"is_phishing": {"type": "noul", "instructions": "Is this email a phishing or scam attempt?"}}
rng = np.random.default_rng(0)
idx = rng.permutation(len(rows)); half = len(rows) // 2
cal_i, test_i = idx[:half], idx[half:]

def auroc(y, p):
    o = np.argsort(p); r = np.empty(len(p), float); r[o] = np.arange(1, len(p)+1)
    n1 = y.sum(); n0 = len(y) - n1
    return (r[y==1].sum() - n1*(n1+1)/2) / (n1*n0)

def fit_platt(p, y, iters=600, lr=0.1):
    z = np.log(np.clip(p,1e-6,1-1e-6) / (1-np.clip(p,1e-6,1-1e-6)))
    a, b = 1.0, 0.0
    for _ in range(iters):
        q = 1/(1+np.exp(-(a*z+b)))
        ga = ((q-y)*z).mean(); gb = (q-y).mean()
        a -= lr*ga; b -= lr*gb
    return a, b

print(f"{'model':<30}{'raw acc':>9}{'fitted acc':>11}{'AUROC':>8}{'best thr':>10}")
for m in args.models:
    agent = laya.load(m) if "/" in m and not m.startswith("/") else laya.Agent(m, device="cuda")
    p = np.array([agent.predict(s, Q)["answers"]["is_phishing"]["noul"] for s, _ in rows])
    y = np.array([lab for _, lab in rows])
    a, b = fit_platt(p[cal_i], y[cal_i])
    z = np.log(np.clip(p,1e-6,1-1e-6)/(1-np.clip(p,1e-6,1-1e-6)))
    q = 1/(1+np.exp(-(a*z+b)))
    raw = ((p[test_i]>=0.5).astype(int) == y[test_i]).mean()
    fit = ((q[test_i]>=0.5).astype(int) == y[test_i]).mean()
    # also the single best raw threshold chosen on the calibration half
    ths = np.unique(np.round(p[cal_i], 4))
    best = max(ths, key=lambda t: ((p[cal_i]>=t).astype(int)==y[cal_i]).mean())
    name = m.split("/")[-2] if m.endswith("ckpt") else m.split("/")[-1]
    print(f"{name:<30}{raw:9.3f}{fit:11.3f}{auroc(y[test_i],p[test_i]):8.3f}{best:10.4f}")
print(f"\n{'TypeSafe Jev (published, raw)':<30}{0.626:9.3f}{'-':>11}{0.689:8.3f}")
